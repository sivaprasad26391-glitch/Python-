data = """station,date,pm25
Adyar,2026-10-01,48
Adyar,2026-10-02,NA

Guindy,2026-10-01,95
Guindy,2026-10-02,-5
Guindy,2026-10-03,121
Adyar,2026-10-04,1200
"""

def clean_data(data):
    result = []
    reader = csv.DictReader(io.StringIO(data))

    for row in reader:
        try:
            station = row["station"].strip().title()
            date = row["date"].strip()
            value = row["pm25"].strip()

            if not station or not date or value in ("", "NA"):
                continue

            pm = float(value)

            if pm < 0 or pm > 999:
                continue

            result.append((station, date, pm))

        except (ValueError, KeyError, TypeError):
            continue

    return result

clean = clean_data(data)
print("TASK B1 - CLEANED DATA")
print(clean)

    for x in values
]

print("Normalized values:", normalized)

# Z-SCORE STANDARDIZATION

mean = statistics.mean(values)
sd = statistics.pstdev(values)

if sd != 0:
    zscore = [round((x - mean) / sd, 3) for x in values]
else:
    zscore = [0 for x in values]

print("Z-score values:", zscore)

# SENTIMENT ADVERSARIAL TESTING

POSITIVE = {"good", "clean", "fast", "helpful", "improved"}
NEGATIVE = {"bad", "dirty", "slow", "rude", "smoke", "broken"}
NEGATORS = {"not", "never", "no"}

def score(text):
    words = re.findall(r"\b\w+\b", text.lower())
    total = 0

    for i, word in enumerate(words):
        value = 1 if word in POSITIVE else -1 if word in NEGATIVE else 0

        if value and i > 0 and words[i - 1] in NEGATORS:
            value = -value

        total += value

    return total

def label(text):
    value = score(text)

    if value > 0:
        return "Positive"
    elif value < 0:
        return "Negative"
    return "Neutral"

]

    print(sentence, ":", label(sentence))

def generate_data(stations):
    groups = {}

    for s in range(stations):
        station = "Station" + str(s + 1)

        for i in range(8760):
            day = "Day" + str(i // 24 + 1)
            key = (station, day)

            if key not in groups:
                groups[key] = [0, 0]

            pm = (i % 100) + 1
            groups[key][0] += pm
            groups[key][1] += 1

    return groups

for count in [1, 12, 120]:
    start = time.perf_counter()
    result = generate_data(count)
    end = time.perf_counter()

    print("Stations:", count)
    print("Readings:", count * 8760)
    print("Groups:", len(result))
    print("Time:", round(end - start, 3), "seconds")

# EXTRACTIVE SUMMARIZER

STOPWORDS = {
    "the", "is", "and", "a", "an",
    "to", "of", "in", "was", "are"
}

def summarise(text, k=2):
    sentences = re.split(r"(?<=[.!?])\s+", text.strip())

    words = [
        w.lower() for w in re.findall(r"\b\w+\b", text)
        if w.lower() not in STOPWORDS
    ]

    frequency = {}

    for word in words:
        frequency[word] = frequency.get(word, 0) + 1

    scores = []

    for sentence in sentences:
        words_in_sentence = re.findall(r"\b\w+\b", sentence)
        value = sum(
            frequency.get(w.lower(), 0)
            for w in words_in_sentence
        )
        scores.append(value)

    selected = sorted(
        range(len(sentences)),
        key=lambda i: scores[i],
        reverse=True
    )[:k]

    return " ".join(
        sentences[i] for i in sorted(selected)
    )

print("\nEXTRACTIVE SUMMARY")

complaint = (
    "The road is damaged. The road has many potholes. "
    "People face problems every day."
)

print(summarise(complaint))

# FIRST K SENTENCES METHOD

def first_k_summary(text, k=2):
    sentences = re.split(r"(?<=[.!?])\s+", text.strip())
    return " ".join(sentences[:k])

print("First K Summary:", first_k_summary(complaint))
